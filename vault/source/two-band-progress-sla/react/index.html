import { slaElapsedPercent, slaState, durationParts } from '../../../lib/kernels/sla';
export function delivery(value: number | null, pending: number, max: number = 100) {
    const percent = max > 0 && value !== null ? Math.max(0, (value / max) * 100) : 0;
    const width = Math.min(100, percent);
    const pendingWidth = max > 0 && value !== null ? Math.max(0, (pending / max) * 100) : 0;
    const over = percent > 100;
    return {
        width,
        pending: Math.min(100 - width, pendingWidth),
        label:
            value === null
                ? '—'
                : over
                  ? 'Over by ' + Number((percent - 100).toFixed(2)) + '%'
                  : Number(percent.toFixed(2)) + '% approved',
        tone: over ? 'warn' : 'neutral',
    };
}
export function fundingCapacity(received: number, committed: number, currency: string) {
    if (!['USD', 'EUR'].includes(currency))
        throw new RangeError('Choose a funding currency to show capacity.');
    const percent = committed > 0 ? (received / committed) * 100 : 0;
    const format = new Intl.NumberFormat('en', {
        minimumFractionDigits: 2,
        maximumFractionDigits: 2,
    });
    return {
        received: format.format(received) + ' ' + currency,
        committed: format.format(committed) + ' ' + currency,
        width: Math.min(100, Math.max(0, percent)),
        over: percent > 100,
        label:
            percent > 100
                ? 'Over by ' + Number((percent - 100).toFixed(2)) + '%'
                : Number(percent.toFixed(2)) + '% received',
    };
}
export function fundingView(received: number, committed: number, currency: string) {
    try {
        return { value: fundingCapacity(received, committed, currency), error: '' };
    } catch (error) {
        return { value: null, error: (error as Error).message };
    }
}
export function drain(elapsed: number, mode: string, locale: string) {
    const end = 48 * 60;
    const ticket = {
        created_at: 0,
        sla_due_at: mode === 'Missing SLA' ? null : end * 60000,
        status: mode,
        resolved_at: mode === 'Closed' ? elapsed * 60000 : null,
    };
    const state = slaState(ticket, elapsed * 60000);
    const parts = durationParts((end - elapsed) * 60000);
    const amount = new Intl.NumberFormat(locale).format(parts.hours || parts.minutes);
    const ar = locale === 'ar';
    const unit = ar ? (parts.hours ? 'ساعة' : 'دقيقة') : parts.hours ? 'h' : 'm';
    const text = ar
        ? state === 'breached'
            ? 'متأخر بمقدار {amount} ' + unit
            : 'متبقي {amount} ' + unit
        : '{amount}' + unit + (state === 'breached' ? ' overdue' : ' left');
    const label =
        state === 'closed'
            ? 'Closed'
            : state === 'unknown'
              ? '—'
              : text.replace('{amount}', amount);
    return { label, state, width: slaElapsedPercent(ticket, elapsed * 60000) };
}
export function startSlaClock(advance: (minutes: number) => void) {
    let observed = performance.now();
    const id = window.setInterval(() => {
        const now = performance.now();
        advance((now - observed) / 60_000);
        observed = now;
    }, 30_000);
    return () => window.clearInterval(id);
}

export default function progress() {
    let stop: (() => void) | undefined;
    return {
        received: 40000,
        committed: 60000,
        currency: 'USD',
        get funding() {
            return fundingView(this.received, this.committed, this.currency);
        },
        approved: 75,
        pending: 12,
        missing: false,
        elapsed: 0,
        mode: 'Open',
        locale: 'en',
        init() {
            this.restartClock();
        },
        restartClock() {
            stop?.();
            if (this.mode === 'Open')
                stop = startSlaClock((minutes) => {
                    this.elapsed += minutes;
                });
        },
        destroy() {
            stop?.();
        },
        get kpi() {
            return delivery(this.missing ? null : this.approved, this.pending);
        },
        get sla() {
            return drain(this.elapsed, this.mode, this.locale);
        },
    };
}

src/lib/kernels/sla.ts

type Instant = string | number;
export interface SlaTicket {
    created_at: Instant | null;
    sla_due_at: Instant | null;
    resolved_at?: Instant | null;
    status?: string;
    priority?: string;
}

const hour = 3_600_000;
const priority = ['critical', 'high', 'medium', 'low'];
const stamp = (time: Instant | null | undefined) =>
    typeof time === 'number' ? time : time ? Date.parse(time) : NaN;
const closed = (ticket: SlaTicket) =>
    ['closed', 'resolved'].includes(ticket.status?.toLowerCase() ?? '');

export function slaState(ticket: SlaTicket, now: Instant) {
    if (closed(ticket)) return 'closed';
    const left = stamp(ticket.sla_due_at) - stamp(now);
    if (!Number.isFinite(left)) return 'unknown';
    if (left < 0) return 'breached';
    return left < 2 * hour ? 'closing' : 'ok';
}

export function slaElapsedPercent(ticket: SlaTicket, now: Instant): number | null {
    const start = stamp(ticket.created_at);
    const budget = stamp(ticket.sla_due_at) - start;
    const elapsed = stamp(closed(ticket) ? ticket.resolved_at : now) - start;
    if (!Number.isFinite(elapsed) || !Number.isFinite(budget) || budget <= 0) return null;
    return Math.min(100, Math.max(2, (elapsed * 100) / budget));
}

export function rankPriority(raw: string | undefined): number {
    const value = raw?.trim().toLowerCase();
    if (value === 'urgent') return 0;
    const index = priority.indexOf(value ?? '');
    return index < 0 ? priority.length : index;
}

export function compareUrgency(a: SlaTicket, b: SlaTicket, now: Instant): number {
    const rank = { breached: 0, closing: 1, ok: 2, unknown: 3, closed: 4 };
    return (
        rank[slaState(a, now)] - rank[slaState(b, now)] ||
        stamp(a.sla_due_at) - stamp(b.sla_due_at) ||
        rankPriority(a.priority) - rankPriority(b.priority) ||
        stamp(a.created_at) - stamp(b.created_at) ||
        0
    );
}

export function durationParts(milliseconds: number) {
    const minutes = Math.floor(Math.abs(milliseconds) / 60_000);
    return { hours: Math.floor(minutes / 60), minutes: minutes % 60 };
}
