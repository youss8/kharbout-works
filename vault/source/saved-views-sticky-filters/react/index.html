import { useRef, useEffect, type ReactNode } from 'react';
import { useKitModel } from '../../../components/vault/useKitModel';
import createViews from './model';
type Model = ReturnType<typeof createViews>;
function HeaderSelect({
    model,
    field,
    label,
    summary,
    children,
}: {
    model: Model;
    field: 'region' | 'status';
    label: string;
    summary: string;
    children: ReactNode;
}) {
    const host = useRef<HTMLDivElement>(null);
    useEffect(() => {
        if (host.current) return model.bindHeader(host.current, field);
    }, [model, field]);
    return (
        <div ref={host} data-search-label={'Find ' + field}>
            <select hidden aria-hidden="true" multiple={field === 'status'}>
                {children}
            </select>
            <button
                data-trigger
                data-header-trigger
                type="button"
                aria-label={label}
                aria-haspopup="dialog"
                aria-expanded="false"
            >
                {label} <small>{summary}</small>
            </button>
        </div>
    );
}

import { useEffect, useState, useSyncExternalStore } from 'react';

/** Both preview stacks observe the same model. Replace arrays and records when updating them. */
export function useKitModel<T extends object>(create: () => T): T {
    const [store] = useState(() => {
        let version = 0;
        const listeners = new Set<() => void>();
        const model = new Proxy(create(), {
            set(target, key, value) {
                if (Object.is(Reflect.get(target, key), value)) return true;
                const written = Reflect.set(target, key, value);
                if (written) {
                    version++;
                    listeners.forEach((notify) => notify());
                }
                return written;
            },
        });
        return {
            model,
            version: () => version,
            subscribe(notify: () => void) {
                listeners.add(notify);
                return () => {
                    listeners.delete(notify);
                };
            },
        };
    });
    useSyncExternalStore(store.subscribe, store.version, store.version);
    useEffect(() => {
        const model = store.model as T & { init?(): void; destroy?(): void };
        model.init?.();
        return () => model.destroy?.();
    }, [store]);
    return store.model;
}

import { mountHeaderPicker } from './header-picker.ts';

type Role = 'Regional HR' | 'Employee';
type Region = '' | 'North' | 'East';
type Sort = 'Ascending' | 'Descending';
type Filters = { status: string[]; region: Region; query: string; sort: Sort };
type View = Filters & { name: string; role: Role };
const viewKey = 'vault-register-views-v1',
    stateKey = 'vault-register-state-v2';
const records = [
    { label: 'Contract A', status: 'Ending', region: 'North' },
    { label: 'Contract B', status: 'Current', region: 'East' },
    { label: 'Contract C', status: 'Restricted', region: 'North' },
    { label: 'Contract D', status: 'Ending', region: 'East' },
];
const roles: Role[] = ['Regional HR', 'Employee'];
const statuses = ['Current', 'Ending', 'Restricted'];
const allowed = (role: Role) =>
    statuses.filter((s) => role === 'Regional HR' || s !== 'Restricted');
function sanitize(raw: Partial<Filters>, role: Role): Filters {
    return {
        status: allowed(role).filter((s) => Array.isArray(raw.status) && raw.status.includes(s)),
        region: raw.region === 'North' || raw.region === 'East' ? raw.region : '',
        query: typeof raw.query === 'string' ? raw.query.trim().slice(0, 100) : '',
        sort: raw.sort === 'Descending' ? 'Descending' : 'Ascending',
    };
}
function read(key: string) {
    try {
        return JSON.parse(localStorage.getItem(key) ?? 'null') as unknown;
    } catch {
        return null;
    }
}
function object(raw: unknown): raw is Record<string, unknown> {
    return !!raw && typeof raw === 'object' && !Array.isArray(raw);
}
function readViews(fallback: View[]): View[] {
    const loaded = read(viewKey);
    if (!Array.isArray(loaded)) return fallback;
    return loaded
        .flatMap((raw) => {
            if (
                !object(raw) ||
                !roles.includes(raw.role as Role) ||
                typeof raw.name !== 'string' ||
                !raw.name.trim()
            )
                return [];
            const role = raw.role as Role;
            return [{ ...sanitize(raw, role), role, name: raw.name.trim().slice(0, 60) }];
        })
        .slice(-40);
}
function readStates(): Partial<Record<Role, Filters>> {
    const stored = read(stateKey);
    if (!object(stored)) return {};
    return Object.fromEntries(
        roles.flatMap((role) =>
            object(stored[role]) ? [[role, sanitize(stored[role] as Filters, role)]] : [],
        ),
    );
}
export default function createViews() {
    let timer: ReturnType<typeof setTimeout> | undefined;
    let memoryOnly = false;
    const cleanups: (() => void)[] = [];
    const headers = new Set<HTMLElement>();
    return {
        role: 'Regional HR' as Role,
        search: '',
        query: '',
        sort: 'Ascending' as Sort,
        region: '' as Region,
        active: [] as string[],
        name: '',
        views: [] as View[],
        notice: '',
        get allowed() {
            return allowed(this.role);
        },
        get scopedViews() {
            return this.views.filter((v) => v.role === this.role);
        },
        get state(): Filters {
            return { status: this.active, region: this.region, sort: this.sort, query: this.query };
        },
        get statusSummary() {
            return this.active.length > 1
                ? this.active[0] + ' +' + (this.active.length - 1)
                : (this.active[0] ?? 'All');
        },
        get rows() {
            const visible = records.filter(
                (r) =>
                    this.allowed.includes(r.status) &&
                    (!this.active.length || this.active.includes(r.status)) &&
                    (!this.region || r.region === this.region) &&
                    r.label.toLowerCase().includes(this.query.toLowerCase()),
            );
            return this.sort === 'Descending' ? visible.reverse() : visible;
        },
        init() {
            this.views = readViews([]);
            this.restore(readStates()[this.role] ?? sanitize({}, this.role));
            const reload = (e: StorageEvent) => {
                if (e.key === viewKey || e.key === null) this.views = readViews([]);
            };
            window.addEventListener('storage', reload);
            cleanups.push(() => window.removeEventListener('storage', reload));
        },
        destroy() {
            clearTimeout(timer);
            for (const stop of cleanups) stop();
            headers.clear();
        },
        restore(state: Filters) {
            clearTimeout(timer);
            this.search = this.query = state.query;
            this.active = [...state.status];
            this.region = state.region;
            this.sort = state.sort;
            for (const h of headers) h.dispatchEvent(new Event('reset-header-draft'));
        },
        persist() {
            if (memoryOnly) return;
            try {
                localStorage.setItem(
                    stateKey,
                    JSON.stringify({ ...readStates(), [this.role]: this.state }),
                );
            } catch {
                memoryOnly = true;
                this.notice = 'Browser storage unavailable. Filters last for this visit.';
            }
        },
        attachHeader(host: HTMLElement, field: 'status' | 'region') {
            cleanups.push(this.bindHeader(host, field));
        },
        bindHeader(host: HTMLElement, field: 'status' | 'region') {
            headers.add(host);
            const remove = mountHeaderPicker(host, {
                read: () => (field === 'status' ? this.active : [this.region]),
                commit: (values) => {
                    if (field === 'status')
                        this.active = this.allowed.filter((v) => values.includes(v));
                    else this.region = sanitize({ region: values[0] as Region }, this.role).region;
                    this.persist();
                },
            });
            const detach = () => {
                headers.delete(host);
                remove();
            };
            return detach;
        },
        type(value: string) {
            this.search = value.slice(0, 100);
            clearTimeout(timer);
            timer = setTimeout(() => {
                this.query = this.search.trim();
                this.persist();
            }, 300);
        },
        sortBy(sort: string) {
            this.sort = sort === 'Descending' ? 'Descending' : 'Ascending';
            this.persist();
        },
        changeRole(next: Role) {
            if (!roles.includes(next)) return;
            this.query = this.search.trim();
            this.persist();
            const current = this.state,
                clean = sanitize(current, next),
                removed = clean.status.length !== current.status.length;
            this.role = next;
            this.restore(readStates()[next] ?? clean);
            this.notice = removed ? 'Removed filters unavailable to this role.' : 'Role changed.';
            this.persist();
        },
        clearFilter(field: 'status' | 'region' | 'query') {
            if (field === 'status') this.active = [];
            else if (field === 'region') this.region = '';
            else {
                clearTimeout(timer);
                this.query = this.search = '';
            }
            this.persist();
        },
        write(views: View[]) {
            this.views = views.slice(-40);
            if (memoryOnly) return;
            try {
                localStorage.setItem(viewKey, JSON.stringify(this.views));
            } catch {
                memoryOnly = true;
                this.notice = 'Browser storage unavailable. Views last for this visit.';
            }
        },
        save() {
            const name = this.name.trim().slice(0, 60);
            if (!name) return;
            const current = memoryOnly ? this.views : readViews(this.views);
            const view = {
                ...sanitize({ ...this.state, query: this.search }, this.role),
                name,
                role: this.role,
            };
            this.notice = 'View saved.';
            this.write([...current.filter((v) => v.name !== name || v.role !== this.role), view]);
            this.name = '';
        },
        apply(view: View) {
            if (view.role !== this.role) return;
            this.restore(sanitize(view, this.role));
            this.persist();
            this.notice = 'View applied.';
        },
        remove(view: View) {
            if (view.role !== this.role) return;
            const current = memoryOnly ? this.views : readViews(this.views);
            this.notice = 'View deleted.';
            this.write(current.filter((v) => v.role !== this.role || v.name !== view.name));
        },
    };
}

type Choice = { value: string; label: string; disabled: boolean };
type PickerOptions = { read: () => readonly string[]; commit: (selected: string[]) => void };
/** Options remain native <option> elements at the call site. Only Done commits the local draft. */
export function mountHeaderPicker(host: HTMLElement, { read, commit }: PickerOptions) {
    const source = host.querySelector('select')!;
    const trigger = host.querySelector<HTMLButtonElement>('[data-trigger]')!;
    const panel = document.createElement('div');
    panel.popover = 'auto';
    panel.className = 'header-options';
    panel.setAttribute('role', 'dialog');
    panel.setAttribute('aria-label', trigger.getAttribute('aria-label') ?? 'Header filter');
    const search = document.createElement('input');
    search.type = 'search';
    search.setAttribute('aria-label', host.dataset.searchLabel ?? 'Find option');
    const list = document.createElement('div');
    list.className = 'header-results';
    const actions = document.createElement('div');
    actions.className = 'kit-row';
    const action = (text: string, run: () => void) => {
        const button = document.createElement('button');
        button.type = 'button';
        button.textContent = text;
        button.addEventListener('click', run);
        actions.append(button);
    };
    let draft: string[] = [],
        options: Choice[] = [];
    const close = () => {
        panel.hidePopover();
        trigger.focus();
    };
    function render(focusValue?: string) {
        list.replaceChildren();
        const matched = options.filter((o) =>
            o.label.toLocaleLowerCase().includes(search.value.trim().toLocaleLowerCase()),
        );
        if (!matched.length) {
            const p = document.createElement('p');
            p.textContent = 'No matching options.';
            list.append(p);
        }
        for (const option of matched) {
            const label = document.createElement('label'),
                input = document.createElement('input');
            input.type = source.multiple ? 'checkbox' : 'radio';
            input.checked = draft.includes(option.value);
            input.disabled = option.disabled;
            input.value = option.value;
            input.addEventListener('change', () => {
                draft = source.multiple
                    ? options
                          .filter((o) =>
                              o.value === option.value ? input.checked : draft.includes(o.value),
                          )
                          .map((o) => o.value)
                    : [option.value];
                render(option.value);
            });
            const text = document.createElement('span');
            text.textContent = option.label;
            label.append(input, text);
            list.append(label);
            if (option.value === focusValue) input.focus();
        }
    }
    function place() {
        const bounds = trigger.getBoundingClientRect();
        const w = window.innerWidth,
            h = window.innerHeight,
            rtl = getComputedStyle(host).direction === 'rtl';
        const start = Math.max(
            6,
            Math.min(rtl ? w - bounds.right : bounds.left, w - panel.offsetWidth - 6),
        );
        panel.style.insetInlineStart = start + 'px';
        const below = h - bounds.bottom - 6;
        panel.style.insetBlockStart =
            (below >= panel.offsetHeight
                ? bounds.bottom
                : Math.max(6, bounds.top - panel.offsetHeight)) + 'px';
    }
    const open = () => {
        if (panel.matches(':popover-open')) {
            close();
            return;
        }
        options = Array.from(source.options, (o) => ({
            value: o.value,
            label: o.label,
            disabled: o.disabled,
        }));
        draft = [...read()];
        search.value = '';
        render();
        panel.showPopover();
        place();
        search.focus();
        trigger.setAttribute('aria-expanded', 'true');
    };
    action('Clear', () => {
        draft = source.multiple ? [] : [source.dataset.default ?? ''];
        render();
        search.focus();
    });
    action('Cancel', close);
    action('Done', () => {
        commit(draft);
        close();
    });
    panel.addEventListener('toggle', () =>
        trigger.setAttribute('aria-expanded', String(panel.matches(':popover-open'))),
    );
    panel.addEventListener('keydown', (e) => {
        if (e.key === 'Escape') {
            e.preventDefault();
            e.stopPropagation();
            close();
            return;
        }
        if (e.key === 'ArrowUp' || e.key === 'ArrowDown') {
            e.preventDefault();
            const enabled = Array.from(
                list.querySelectorAll<HTMLInputElement>('input:not(:disabled)'),
            );
            if (!enabled.length) return;
            const index = enabled.indexOf(document.activeElement as HTMLInputElement);
            const dest =
                index < 0
                    ? e.key === 'ArrowDown'
                        ? 0
                        : enabled.length - 1
                    : (index + (e.key === 'ArrowDown' ? 1 : -1) + enabled.length) % enabled.length;
            enabled[dest].focus();
        } else if (e.target === search && e.key.length === 1) {
            e.stopPropagation();
        }
    });
    search.addEventListener('input', () => render());
    panel.append(search, list, actions);
    host.append(panel);
    trigger.addEventListener('click', open);
    const reposition = () => {
        if (panel.matches(':popover-open')) place();
    };
    window.addEventListener('resize', reposition);
    window.addEventListener('scroll', reposition, true);
    const dismiss = () => {
        if (panel.matches(':popover-open')) panel.hidePopover();
    };
    host.addEventListener('reset-header-draft', dismiss);
    return () => {
        trigger.removeEventListener('click', open);
        host.removeEventListener('reset-header-draft', dismiss);
        window.removeEventListener('resize', reposition);
        window.removeEventListener('scroll', reposition, true);
        panel.remove();
    };
}
