<?php

namespace App\Livewire;

use App\Livewire\Concerns\HasSavedViews;
use Illuminate\Support\Facades\Gate;
use Livewire\Attributes\Locked;
use Livewire\Component;

class SavedRegisterViews extends Component
{
    use HasSavedViews;

    public string $role = 'Employee';

    public string $query = '';

    public string $name = '';

    public string $sort = 'Ascending';

    public string $notice = '';

    #[Locked]
    public string $region = '';

    #[Locked]
    public array $status = [];

    public array $draft = [];

    public function allowed(): array
    {
        return $this->role === 'Regional HR' && Gate::allows('vault.regional-hr') ? ['Current', 'Ending', 'Restricted'] : ['Current', 'Ending'];
    }

    public function mount(): void
    {
        $this->role = Gate::allows('vault.regional-hr') ? 'Regional HR' : 'Employee';
        $this->restoreTable(session('example.register.filters.'.$this->role, []));
    }

    public function updatedRole(): void
    {
        $this->validate(['role' => 'in:Regional HR,Employee']);
        if ($this->role === 'Regional HR' && ! Gate::allows('vault.regional-hr')) {
            $this->role = 'Employee';
        }
        $current = $this->tableState();
        $trimmed = count(array_diff($current['status'], $this->allowed())) > 0;
        $this->restoreTable(session('example.register.filters.'.$this->role, $current));
        $this->persistFilters();
        $this->notice = $trimmed ? 'Removed filters unavailable to this role.' : 'Role changed.';
    }

    public function updatedQuery(): void
    {
        $this->query = mb_substr(trim($this->query), 0, 100);
        $this->persistFilters();
    }

    public function updatedSort(): void
    {
        $this->validate(['sort' => 'in:Ascending,Descending']);
        $this->persistFilters();
    }

    public function applyFilters(?array $draft = null): void
    {
        $requested = $draft ?? $this->draft;
        $clean = $this->restoreTable([...$this->tableState(), 'status' => $requested]);
        $this->persistFilters();
        $this->notice = $clean ? 'Removed filters unavailable to this role.' : 'Filters applied.';
    }

    public function applyRegion(string $region): void
    {
        abort_unless(in_array($region, ['', 'North', 'East'], true), 422);
        $this->region = $region;
        $this->persistFilters();
    }

    public function render()
    {
        $rows = collect([
            ['Contract A', 'North', 'Ending'], ['Contract B', 'East', 'Current'],
            ['Contract C', 'North', 'Restricted'], ['Contract D', 'East', 'Ending'],
        ])->filter(fn ($row) => in_array($row[2], $this->allowed(), true)
            && (! $this->status || in_array($row[2], $this->status, true)) && (! $this->region || $row[1] === $this->region)
            && str_contains(mb_strtolower($row[0]), mb_strtolower($this->query)));

        return view('livewire.saved-views-sticky-filters', ['rows' => $this->sort === 'Descending' ? $rows->reverse() : $rows, 'views' => $this->roleViews()]);
    }
}

<?php

namespace App\Livewire\Concerns;

trait HasSavedViews
{
    private function roleViews(): array
    {
        return array_filter(session('example.register.views', []), fn ($view) => is_array($view) && ($view['role'] ?? null) === $this->role);
    }

    private function tableState(): array
    {
        return ['status' => $this->status, 'region' => $this->region, 'sort' => $this->sort, 'query' => $this->query];
    }

    private function restoreTable(array $state): bool
    {
        $requested = array_filter((array) ($state['status'] ?? []), 'is_string');
        $this->status = array_values(array_intersect($this->allowed(), $requested));
        $this->region = in_array($state['region'] ?? '', ['North', 'East'], true) ? $state['region'] : '';
        $this->query = mb_substr(trim(is_string($state['query'] ?? null) ? $state['query'] : ''), 0, 100);
        $this->sort = ($state['sort'] ?? '') === 'Descending' ? 'Descending' : 'Ascending';
        $this->draft = $this->status;

        return count($requested) !== count($this->status);
    }

    private function persistFilters(): void
    {
        session(['example.register.filters.'.$this->role => $this->tableState()]);
    }

    public function saveView(): void
    {
        $this->name = trim($this->name);
        $this->validate(['name' => 'required|string|max:60', 'query' => 'nullable|string|max:100', 'sort' => 'in:Ascending,Descending', 'region' => 'nullable|in:North,East', 'role' => 'in:Regional HR,Employee']);
        $views = session('example.register.views', []);
        $views[hash('sha256', $this->role.'|'.$this->name)] = ['name' => $this->name, 'role' => $this->role, ...$this->tableState()];
        session(['example.register.views' => array_slice($views, -40, null, true)]);
        $this->name = '';
        $this->notice = 'View saved.';
    }

    public function applyView(string $id): void
    {
        $view = $this->roleViews()[$id] ?? abort(404);
        $this->notice = $this->restoreTable($view) ? 'Removed filters unavailable to this role.' : 'View applied.';
        $this->persistFilters();
    }

    public function deleteView(string $id): void
    {
        abort_unless(isset($this->roleViews()[$id]), 404);
        $views = session('example.register.views', []);
        unset($views[$id]);
        session(['example.register.views' => $views]);
        $this->notice = 'View deleted.';
    }
}
