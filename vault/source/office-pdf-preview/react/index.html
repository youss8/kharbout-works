                        rel="noopener noreferrer"
                        href={m.kind === 'pdf' ? m.current.url : m.standaloneUrl}
                    >
                        Open preview in new tab
                    </a>
                )}
                <a href={m.current.url} download={m.current.name}>
                    Download original
                </a>
            </dialog>
        </section>
    );
}

src/components/vault/useKitModel.ts

import { useEffect, useState, useSyncExternalStore } from 'react';

/** Both preview stacks observe the same model. Replace arrays and records when updating them. */
export function useKitModel<T extends object>(create: () => T): T {
    const [store] = useState(() => {
        let version = 0;
        const listeners = new Set<() => void>();
        const model = new Proxy(create(), {
            set(target, key, value) {
                if (Object.is(Reflect.get(target, key), value)) return true;
                const written = Reflect.set(target, key, value);
                if (written) {
                    version++;
                    listeners.forEach((notify) => notify());
                }
                return written;
            },
        });
        return {
            model,
            version: () => version,
            subscribe(notify: () => void) {
                listeners.add(notify);
                return () => {
                    listeners.delete(notify);
                };
            },
        };
    });
    useSyncExternalStore(store.subscribe, store.version, store.version);
    useEffect(() => {
        const model = store.model as T & { init?(): void; destroy?(): void };
        model.init?.();
        return () => model.destroy?.();
    }, [store]);
    return store.model;
}

src/content/vault/office-pdf-preview/model.ts

import workbookUrl from './sample.xlsx?url';
import documentUrl from './sample.docx?url';
import pdfUrl from './sample.pdf?url';
import archiveUrl from './sample.zip?url';
import type { readWorkbook } from './office-reader';
type Workbook = Awaited<ReturnType<typeof readWorkbook>>;
type FileEntry = { name: string; url: string; file?: File };
const samples: FileEntry[] = [
    { name: 'q3-budget.xlsx', url: workbookUrl },
    { name: 'grant-brief.docx', url: documentUrl },
    { name: 'review.pdf', url: pdfUrl },
    { name: 'archive.zip', url: archiveUrl },
];
export default function createViewer() {
    let version = 0,
        workbook: Workbook | null = null;
    const ownedUrls: string[] = [];
    return {
        files: [...samples],
        index: 0,
        sheet: 0,
        sheets: [] as string[],
        rows: [] as string[][],
        documentHtml: '',
        standaloneUrl: '',
        totalRows: 0,
        totalCols: 0,
        kind: '',
        error: '',
        loading: false,
        opened: false,
        get current() {
            return this.files[this.index];
        },
        get clipped() {
            return this.totalRows > 500 || this.totalCols > 50;
        },
        async open() {
            this.opened = true;
            document.querySelector<HTMLDialogElement>('#office-modal')!.showModal();
            await this.load(this.index);
        },
        close() {
            this.opened = false;
            version++;
            this.loading = false;
            document.querySelector<HTMLDialogElement>('#office-modal')?.close();
        },
        async add(list: FileList | null) {
            if (!list?.length) return;
            const added = Array.from(list).map((file) => {
                const content = /\.pdf$/i.test(file.name)
                    ? new Blob([file], { type: 'application/pdf' })
                    : file;
                const url = URL.createObjectURL(content);
                ownedUrls.push(url);
                return { name: file.name, url, file };
            });
            const index = this.files.length;
            this.files = [...this.files, ...added];
            await this.load(index);
        },
        async load(index: number) {
            const file = this.files[index];
            if (!file) return;
            const token = ++version;
            this.index = index;
            this.loading = true;
            this.error = '';
            this.kind = '';
            this.sheet = 0;
            this.rows = [];
            if (this.standaloneUrl) URL.revokeObjectURL(this.standaloneUrl);
            this.standaloneUrl = '';
            this.documentHtml = '';
            this.sheets = [];
            this.totalRows = this.totalCols = 0;
            workbook = null;
            const ext = file.name.split('.').at(-1)?.toLowerCase();
            try {
                if (ext === 'pdf') {
                    this.kind = 'pdf';
                    return;
                }
                if (ext !== 'xlsx' && ext !== 'docx') {
                    this.kind = 'unsupported';
                    return;
                }
                if (file.file && file.file.size > 20 * 1024 * 1024)
                    throw new Error('Office files are limited to 20 MB.');
                const getData = async () => {
                    if (file.file) return file.file.arrayBuffer();
                    const res = await fetch(file.url);
                    if (!res.ok) throw new Error('Sample file could not load.');
                    return res.arrayBuffer();
                };
                const [reader, data] = await Promise.all([import('./office-reader'), getData()]);
                if (token !== version) return;
                if (ext === 'xlsx') {
                    const book = await reader.readWorkbook(data);
                    const first = await book.read(0);
                    if (token !== version) return;
                    workbook = book;
                    this.sheets = book.sheets;
                    Object.assign(this, first);
                    this.kind = 'sheet';
                    this.updateStandalone();
                } else {
                    const doc = await reader.readDocument(data);
                    if (token !== version) return;
                    this.documentHtml = this.page(doc);
                    this.kind = 'document';
                    this.updateStandalone();
                }
            } catch (err) {
                if (token === version)
                    this.error = err instanceof Error ? err.message : 'File could not be read.';
            } finally {
                if (token === version) this.loading = false;
            }
        },
        moveSheet(event: KeyboardEvent, list: HTMLElement) {
            const rtl = getComputedStyle(list).direction === 'rtl';
            let index = this.sheet;
            if (event.key === 'Home') index = 0;
            else if (event.key === 'End') index = this.sheets.length - 1;
            else if (['ArrowLeft', 'ArrowRight'].includes(event.key)) {
                const step = (event.key === 'ArrowRight') !== rtl ? 1 : -1;
                index = (index + step + this.sheets.length) % this.sheets.length;
            } else return;
            event.preventDefault();
            list.querySelectorAll<HTMLButtonElement>('[role=tab]')[index]?.focus();
            void this.chooseSheet(index);
        },
        async chooseSheet(index: number) {
            if (!workbook) return;
            const token = ++version;
            this.sheet = index;
            this.loading = true;
            this.error = '';
            try {
                const result = await workbook.read(index);
                if (token === version) {
                    Object.assign(this, result);
                    this.updateStandalone();
                }
            } catch (err) {
                if (token === version)
                    this.error = err instanceof Error ? err.message : 'Worksheet unavailable.';
            } finally {
                if (token === version) this.loading = false;
            }
        },
        page(markup: string) {
            return (
                '<!doctype html><html><head><meta charset="utf-8"><meta http-equiv="Content-Security-Policy" content="default-src &apos;none&apos;; style-src &apos;unsafe-inline&apos;; img-src data:; form-action &apos;none&apos;; base-uri &apos;none&apos;"><meta name="viewport" content="width=device-width,initial-scale=1"><title>Local file preview</title><style>body{font:1rem system-ui;margin:0}td{padding:.4rem;border:1px solid #bbb}.docx-wrapper{padding:.4rem!important}.docx-wrapper>section.docx{max-width:100%;padding:1rem!important;box-sizing:border-box}</style></head><body>' +
                markup +
                '</body></html>'
            );
        },
        updateStandalone() {
            if (this.standaloneUrl) URL.revokeObjectURL(this.standaloneUrl);
            let html = this.documentHtml;
            if (this.kind === 'sheet') {
                const table = document.createElement('table');
                for (const row of this.rows) {
                    const tr = table.insertRow();
                    for (const cell of row) tr.insertCell().textContent = String(cell);
                }
                html = this.page(table.outerHTML);
            }
            this.standaloneUrl = URL.createObjectURL(
                new Blob([html], { type: 'text/html;charset=utf-8' }),
            );
        },
        destroy() {
            if (this.standaloneUrl) URL.revokeObjectURL(this.standaloneUrl);
            version++;
            ownedUrls.forEach((url) => URL.revokeObjectURL(url));
        },
    };
}

src/content/vault/office-pdf-preview/office-reader.ts

const decoder = new TextDecoder();
const maxPart = 12 * 1024 * 1024;
// Check the archive budget before passing the file to either document parser.
function checkPackage(buffer: ArrayBuffer, requiredPart: string) {
    const view = new DataView(buffer);
    let end = buffer.byteLength - 22;
    const limit = Math.max(0, buffer.byteLength - 65557);
    while (end >= limit && view.getUint32(end, true) !== 0x06054b50) end--;
    if (end < limit) throw new Error('Missing Office ZIP directory.');
    let at = view.getUint32(end + 16, true);
    const count = view.getUint16(end + 10, true);
    let total = 0,
        found = false;
    if (count > 4096) throw new Error('This document contains too many parts.');
    for (let i = 0; i < count; i++) {
        if (at + 46 > end || view.getUint32(at, true) !== 0x02014b50)
            throw new Error('Invalid Office directory.');
        const length = view.getUint16(at + 28, true);
        const size = view.getUint32(at + 24, true);
        total += size;
        if (size > maxPart || total > 48 * 1024 * 1024)
            throw new Error('Document expansion limit exceeded.');
        const name = decoder.decode(new Uint8Array(buffer, at + 46, length));
        found ||= name === requiredPart;
        at += 46 + length + view.getUint16(at + 30, true) + view.getUint16(at + 32, true);
    }
    if (!found) throw new Error('The selected file is missing its Office document part.');
}

export async function readWorkbook(data: ArrayBuffer) {
    checkPackage(data, 'xl/workbook.xml');
    const { read, utils } = await import('xlsx');
    const book = read(data, { cellHTML: false, cellFormula: false, cellStyles: false });
    if (!book.SheetNames.length) throw new Error('The workbook has no sheets.');
    return {
        sheets: book.SheetNames,
        async read(index: number) {
            const sheet = book.Sheets[book.SheetNames[index]];
            if (!sheet) throw new Error('Unknown worksheet.');
            if (!sheet['!ref']) return { rows: [], totalRows: 0, totalCols: 0 };
            const { e } = utils.decode_range(sheet['!ref']);
            const totalRows = e.r + 1,
                totalCols = e.c + 1;
            const range = { s: { r: 0, c: 0 }, e: { r: Math.min(e.r, 499), c: Math.min(e.c, 49) } };
            const rows = utils.sheet_to_json<string[]>(sheet, {
                header: 1,
                range,
                raw: false,
                defval: '',
                blankrows: true,
            });
            return { rows, totalRows, totalCols };
        },
    };
}

export async function readDocument(data: ArrayBuffer) {
    checkPackage(data, 'word/document.xml');
    const { parseAsync, renderDocument } = await import('docx-preview');
    const options = {
        renderAltChunks: false,
        ignoreFonts: true,
        useBase64URL: true,
        breakPages: true,
        ignoreLastRenderedPageBreak: false,
        ignoreWidth: true,
    };
    const documentModel = await parseAsync(data, options);
    if (!documentModel.documentPart?.body) throw new Error('The document body is missing.');
    const template = document.createElement('template');
    template.content.append(...(await renderDocument(documentModel, options)));
    // Keep passive document content. The preview frame also denies scripts, navigation and network access.
    template.content
        .querySelectorAll('iframe,object,embed,script,link,base,form,foreignObject,meta')
        .forEach((el) => el.remove());
    for (const el of template.content.querySelectorAll('*')) {
        for (const attribute of Array.from(el.attributes)) {
            const key = attribute.name.toLowerCase();
            const safeImage =
                key === 'src' &&
                el.localName === 'img' &&
                /^data:image\/(png|jpeg|gif|webp);base64,/i.test(attribute.value);
            if (
                key.startsWith('on') ||
                (['href', 'xlink:href', 'src', 'srcset', 'action', 'target'].includes(key) &&
                    !safeImage)
            )
                el.removeAttribute(attribute.name);
        }
    }
    return template.innerHTML;
}
