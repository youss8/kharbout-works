import { useReducer, useRef, useState, type CSSProperties } from 'react';
import grid from './model';
import './style.css';

export default function Example() {
    const [desk] = useState(grid),
        [, refresh] = useReducer((n) => n + 1, 0);
    const root = useRef<HTMLDivElement>(null);
    function act(fn: () => void) {
        fn();
        refresh();
    }
    return (
        <section
            className="kit-panel"
            onKeyDown={(e) => {
                if (e.key === 'Escape' && desk.editing) act(() => desk.cancel());
            }}
        >
            <div className="kit-row">
                {desk.editing ? (
                    <>
                        <button onClick={() => act(() => desk.save())}>Done</button>
                        <button onClick={() => act(() => desk.cancel())}>Cancel</button>
                        <button onClick={() => act(() => desk.reset())}>
                            Reset to role default
                        </button>
                    </>
                ) : (
                    <button onClick={() => act(() => desk.customize())}>Customize grid</button>
                )}
            </div>
            {desk.editing && (
                <p>Drag a header or use arrow keys to swap. Shift + arrow resizes a panel.</p>
            )}
            <div ref={root} className="desk-layout">
                {desk.rows.flat().map((p) => (
                    <article
                        key={p.panel}
                        data-grid-panel
                        style={{ '--span': p.cols } as CSSProperties}
                        onDragOver={(e) => e.preventDefault()}
                        onDrop={(e) => {
                            e.preventDefault();
                            act(() => desk.swap(desk.drag, p.panel, root.current!));
                        }}
                    >
                        <header>
                            <button
                                aria-label={'Move ' + p.title}
                                disabled={!desk.editing}
                                draggable={desk.editing}
                                onDragStart={(e) => {
                                    desk.drag = p.panel;
                                    e.dataTransfer.setData('text/plain', p.panel);
                                }}
                                onKeyDown={(e) =>
                                    act(() => desk.keys(e.nativeEvent, p.panel, root.current!))
                                }
                            >
                                {p.title}
                            </button>
                            <bdi>{p.cols}/12</bdi>
                        </header>
                        {desk.editing && (
                            <div className="kit-row">
                                <button
                                    disabled={!desk.canSize(p.panel, -2)}
                                    aria-label={'Shrink ' + p.title}
                                    onClick={() => act(() => desk.size(p.panel, -2, root.current!))}
                                >
                                    −2
                                </button>
                                <button
                                    disabled={!desk.canSize(p.panel, 2)}
                                    aria-label={'Grow ' + p.title}
                                    onClick={() => act(() => desk.size(p.panel, 2, root.current!))}
                                >
                                    +2
                                </button>
                                <button
                                    className="grip"
                                    aria-label={'Drag to resize ' + p.title}
                                    onPointerDown={(e) =>
                                        desk.edgeStart(
                                            e.nativeEvent,
                                            p.panel,
                                            root.current!,
                                            e.currentTarget,
                                            refresh,
                                        )
                                    }
                                    onKeyDown={(e) =>
                                        act(() => desk.keys(e.nativeEvent, p.panel, root.current!))
                                    }
                                >
                                    ↔
                                </button>
                            </div>
                        )}
                    </article>
                ))}
            </div>
            <output>{desk.monitor}</output>
            <p role="status">{desk.notice}</p>
        </section>
    );
}

src/content/vault/flush-row-desk-grid/model.ts

import { cleanLayout, flushRows, resizePanel } from '../../../lib/kernels/flushRows';

export const catalog = [
    { panel: 'returns', title: 'Monthly returns', cols: 6, rows: 2, minCols: 4 },
    { panel: 'funding', title: 'Funding health', cols: 6, rows: 2, minCols: 4 },
    { panel: 'queue', title: 'Priority queue', cols: 8, rows: 2, minCols: 4 },
    { panel: 'digest', title: 'Needs action', cols: 4, rows: 2, minCols: 4 },
];
const storageKey = 'vault.operations-desk.v1';
function readLayout() {
    try {
        return cleanLayout(JSON.parse(localStorage.getItem(storageKey) ?? 'null'), catalog);
    } catch {
        return cleanLayout(null, catalog);
    }
}
function flip(root: Element, update: () => void) {
    const panels = Array.from(root.querySelectorAll<HTMLElement>('[data-grid-panel]'));
    const old = panels.map((el) => [el, el.getBoundingClientRect()] as const);
    update();
    if (matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    requestAnimationFrame(() => {
        for (const [el, before] of old) {
            const after = el.getBoundingClientRect();
            if (before.x === after.x && before.y === after.y) continue;
            el.animate(
                [
                    {
                        transform:
                            'translate(' +
                            (before.x - after.x) +
                            'px,' +
                            (before.y - after.y) +
                            'px)',
                    },
                    { transform: 'none' },
                ],
                { duration: 220, easing: 'cubic-bezier(.25,.8,.25,1)' },
            );
        }
    });
}
export default function deskGrid() {
    return {
        saved: readLayout(),
        layout: readLayout(),
        editing: false,
        drag: '',
        pending: '',
        notice: '',
        get rows() {
            return flushRows(this.layout);
        },
        get monitor() {
            const valid = this.rows.every((row) => row.reduce((sum, p) => sum + p.cols, 0) === 12);
            return (
                this.rows.length +
                ' rows · ' +
                (valid ? 'Every row totals 12 · Zero holes ✓' : 'Check row widths')
            );
        },
        customize() {
            this.layout = cleanLayout(this.saved, catalog);
            this.editing = true;
            this.notice = '';
        },
        cancel() {
            this.layout = cleanLayout(this.saved, catalog);
            this.editing = false;
            this.drag = '';
        },
        save() {
            this.saved = cleanLayout(this.layout, catalog);
            this.layout = this.saved;
            this.editing = false;
            try {
                localStorage.setItem(storageKey, JSON.stringify(this.saved));
                this.notice = 'Layout saved.';
            } catch {
                this.notice = 'Layout saved for this visit. Browser storage unavailable.';
            }
        },
        reset() {
            if (this.editing) this.layout = cleanLayout(null, catalog);
        },
        swap(from: string, to: string, root: Element) {
            if (!this.editing || this.pending || from === to) return;
            const items = this.layout.slice(),
                a = items.findIndex((p) => p.panel === from),
                b = items.findIndex((p) => p.panel === to);
            if (a < 0 || b < 0) return;
            this.pending = from;
            [items[a], items[b]] = [
                { ...items[b], cols: items[a].cols },
                { ...items[a], cols: items[b].cols },
            ];
            flip(root, () => {
                this.layout = flushRows(items).flat();
            });
            this.drag = '';
            queueMicrotask(() => {
                this.pending = '';
            });
        },
        size(id: string, delta: number, root: Element) {
            if (!this.editing) return;
            const row = this.rows.find((group) => group.some((p) => p.panel === id));
            const edge = row?.at(-1)?.panel === id ? 'start' : 'end';
            flip(root, () => {
                this.layout = resizePanel(this.layout, id, edge, delta);
            });
        },
        canSize(id: string, delta: number) {
            const row = this.rows.find((group) => group.some((p) => p.panel === id));
            if (!row || row.length < 2) return false;
            const index = row.findIndex((p) => p.panel === id),
                panel = row[index],
                neighbor = row[index + 1] ?? row[index - 1];
            return delta < 0 ? panel.cols > panel.minCols : neighbor.cols > neighbor.minCols;
        },
        keys(e: KeyboardEvent, id: string, root: Element) {
            if (e.key === 'Escape') {
                this.cancel();
                return;
            }
            if (
                !['ArrowLeft', 'ArrowRight', 'ArrowUp', 'ArrowDown'].includes(e.key) ||
                !this.editing
            )
                return;
            e.preventDefault();
            let delta = e.key === 'ArrowRight' || e.key === 'ArrowDown' ? 1 : -1;
            if (
                (e.key === 'ArrowRight' || e.key === 'ArrowLeft') &&
                getComputedStyle(root).direction === 'rtl'
            )
                delta *= -1;
            if (e.shiftKey) this.size(id, delta * 2, root);
            else {
                const target = this.layout[this.layout.findIndex((p) => p.panel === id) + delta];
                if (target) this.swap(id, target.panel, root);
            }
        },
        edgeStart(
            e: PointerEvent,
            id: string,
            root: Element,
            grip = e.currentTarget as HTMLElement,
            refresh: () => void = () => {},
        ) {
            if (!this.editing || e.button !== 0) return;
            const start = e.clientX,
                cell = root.getBoundingClientRect().width / 12;
            const row = this.rows.find((group) => group.some((p) => p.panel === id));
            const sign =
                (getComputedStyle(root).direction === 'rtl' ? -1 : 1) *
                (row?.at(-1)?.panel === id ? -1 : 1);
            const baseline = this.layout.slice(),
                edge = row?.at(-1)?.panel === id ? 'start' : 'end';
            const initial = baseline.find((p) => p.panel === id)!.cols;
            let applied = 0;
            const move = (event: PointerEvent) => {
                if (event.pointerId !== e.pointerId) return;
                const delta = ((event.clientX - start) * sign) / cell - applied;
                if (Math.abs(delta) < 0.65) return;
                const requested = applied + Math.sign(delta) * Math.floor(Math.abs(delta) + 0.35);
                const next = resizePanel(baseline, id, edge, requested);
                const newStep = next.find((p) => p.panel === id)!.cols - initial;
                if (newStep === applied) return;
                applied = newStep;
                this.layout = next;
                refresh();
            };
            const cleanup = () => {
                grip.removeEventListener('pointermove', move);
                grip.removeEventListener('pointerup', finish);
                grip.removeEventListener('pointercancel', cancel);
                grip.removeEventListener('lostpointercapture', cancel);
            };
            const finish = (event: PointerEvent) => {
                move(event);
                cleanup();
            };
            const cancel = () => {
                cleanup();
                this.layout = baseline;
                refresh();
            };
            grip.setPointerCapture(e.pointerId);
            grip.addEventListener('pointermove', move);
            grip.addEventListener('pointerup', finish);
            grip.addEventListener('pointercancel', cancel);
            grip.addEventListener('lostpointercapture', cancel);
        },
    };
}

src/lib/kernels/flushRows.ts

export const gridColumns = 12;

export interface GridPanel {
    panel: string;
    cols: number;
    minCols?: number;
}

function span(value: number | undefined, fallback: number): number {
    if (value === undefined || !Number.isFinite(value)) return fallback;
    return Math.min(gridColumns, Math.max(1, Math.round(value)));
}

/** Pack in reading order. The last panel in each row absorbs any spare columns. */
export function flushRows<T extends GridPanel>(layout: readonly T[]): T[][] {
    const rows: T[][] = [];
    let row: T[] = [];
    let used = 0;
    function finishRow() {
        if (!row.length) return;
        row[row.length - 1].cols += gridColumns - used;
        rows.push(row);
        row = [];
        used = 0;
    }
    for (const panel of layout) {
        const min = span(panel.minCols, 1);
        const cols = Math.max(min, span(panel.cols, min));
        if (used + cols > gridColumns) finishRow();
        row.push({ ...panel, cols });
        used += cols;
    }
    finishRow();
    return rows;
}

/** Accept only known panel IDs and widths. The current catalog owns all other fields. */
export function cleanLayout<T extends GridPanel>(stored: unknown, catalog: readonly T[]): T[] {
    const remaining = new Map(catalog.map((panel) => [panel.panel, panel]));
    const result: T[] = [];
    if (Array.isArray(stored)) {
        for (const entry of stored) {
            if (!entry || typeof entry.panel !== 'string') continue;
            const recipe = remaining.get(entry.panel);
            if (!recipe) continue;
            const min = span(recipe.minCols, 1);
            const width = typeof entry.cols === 'number' ? entry.cols : undefined;
            result.push({ ...recipe, cols: Math.max(min, span(width, span(recipe.cols, min))) });
            remaining.delete(entry.panel);
        }
    }
    return flushRows([...result, ...remaining.values()]).flat();
}

/** A positive delta grows the panel by borrowing from the neighbor at the chosen edge. */
export function resizePanel<T extends GridPanel>(
    layout: readonly T[],
    id: string,
    edge: 'start' | 'end',
    delta: number,
): T[] {
    const rows = flushRows(layout);
    if (!Number.isFinite(delta)) return rows.flat();
    for (const row of rows) {
        const index = row.findIndex((panel) => panel.panel === id);
        if (index < 0) continue;
        const neighbor = row[index + (edge === 'start' ? -1 : 1)];
        if (!neighbor) break;
        const panel = row[index];
        const applied = Math.max(
            span(panel.minCols, 1) - panel.cols,
            Math.min(neighbor.cols - span(neighbor.minCols, 1), Math.round(delta)),
        );
        panel.cols += applied;
        neighbor.cols -= applied;
        break;
    }
    return rows.flat();
}
