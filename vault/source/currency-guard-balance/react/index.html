import { useState } from 'react';
import { deriveBalance, ledgerSummary } from './model';
export default function MoneyBalance() {
    const [used, setUsed] = useState(40),
        [credit, setCredit] = useState(true),
        [pending, setPending] = useState(true),
        [mix, setMix] = useState(false);
    const [eligible, setEligible] = useState(true);
    const ledger = ledgerSummary(mix),
        left = deriveBalance(used, credit, pending);
    return (
        <section className="kit-panel">
            <label>
                <input type="checkbox" checked={mix} onChange={(e) => setMix(e.target.checked)} />{' '}
                Attempt mixed-currency total
            </label>
            {ledger.reason && (
                <div className="error-card" role="alert">
                    {ledger.reason}
                    <button onClick={() => setMix(false)}>Retry per currency</button>
                </div>
            )}
            <div className="kit-grid">
                {ledger.totals.map((total) => (
                    <bdi className="kit-cell" key={total}>
                        {total}
                    </bdi>
                ))}
            </div>
            <label>
                <input
                    type="checkbox"
                    checked={eligible}
                    onChange={(e) => setEligible(e.target.checked)}
                />{' '}
                Annual leave entitlement
            </label>
            {!eligible ? (
                <p>This contract has no annual leave entitlement.</p>
            ) : (
                <>
                    <label>
                        Used hours
                        <input
                            type="range"
                            min="0"
                            max="160"
                            step=".25"
                            value={used}
                            onChange={(e) => setUsed(e.target.valueAsNumber)}
                        />
                    </label>
                    <div className="kit-row">
                        <label>
                            <input
                                type="checkbox"
                                checked={credit}
                                onChange={(e) => setCredit(e.target.checked)}
                            />{' '}
                            Include 8h overtime credit
                        </label>
                        <label>
                            <input
                                type="checkbox"
                                checked={pending}
                                onChange={(e) => setPending(e.target.checked)}
                            />{' '}
                            Reserve 16h pending
                        </label>
                    </div>
                    <p>
                        120h entitlement {credit && '+ 8h credit'} − <bdi>{used}</bdi>h used{' '}
                        {pending && '− 16h pending'}
                    </p>
                    <div
                        role="img"
                        className="balance-bands"
                        aria-label={
                            'Used ' +
                            used +
                            ' hours, reserved ' +
                            (pending ? 16 : 0) +
                            ' hours, remaining ' +
                            left.hours +
                            ' hours'
                        }
                    >
                        <i className="used" style={{ width: left.usedWidth + '%' }} />
                        <i className="reserved" style={{ width: left.pendingWidth + '%' }} />
                        {credit && (
                            <i
                                className="credit"
                                style={{ insetInlineStart: left.creditStart + '%' }}
                            />
                        )}
                    </div>
                    <output className="kit-cell">
                        {left.hours < 0 && <span>△ Overdrawn</span>}
                        <span>Remaining balance</span>
                        <bdi className="kit-value">{left.duration}</bdi>
                        <span>{left.hours} hours</span>
                    </output>
                </>
            )}
        </section>
    );
}

export const ledger = [
    { amount: 125000.5, currency: 'EUR' },
    { amount: 420000, currency: 'USD' },
];
export function formatMoney(amount: number, currency: string | null | undefined) {
    if (!currency?.trim()) throw new Error('A currency code is required.');
    const code = currency.trim().toUpperCase();
    const parts = new Intl.NumberFormat('en', {
        style: 'currency',
        currency: code,
        currencyDisplay: 'code',
    }).formatToParts(amount);
    return (
        parts
            .filter((part) => part.type !== 'currency' && part.type !== 'literal')
            .map((part) => part.value)
            .join('') +
        ' ' +
        code
    );
}
export function subtotal(rows: typeof ledger) {
    if (!rows.length) throw new Error('Choose at least one ledger entry.');
    const currencies = new Set(rows.map((row) => row.currency));
    if (currencies.size !== 1)
        throw new Error('Select a single currency. The ledger contains EUR and USD.');
    return formatMoney(
        rows.reduce((sum, r) => sum + r.amount, 0),
        rows[0].currency,
    );
}
export function ledgerSummary(mix: boolean) {
    try {
        return {
            totals: mix ? [subtotal(ledger)] : ledger.map((row) => subtotal([row])),
            reason: '',
        };
    } catch (error) {
        return { totals: ledger.map((row) => subtotal([row])), reason: (error as Error).message };
    }
}
export function deriveBalance(used: number, credit: boolean, pending: boolean) {
    const capacity = 120 + (credit ? 8 : 0);
    const remaining = capacity - used - (pending ? 16 : 0);
    const width = Math.max(0, Math.min(100, (100 * used) / capacity));
    const minutes = Math.round(Math.abs(remaining) * 60);
    return {
        hours: remaining,
        usedWidth: width,
        pendingWidth: Math.min(100 - width, (100 * (pending ? 16 : 0)) / capacity),
        creditStart: (100 * 120) / capacity,
        duration:
            (remaining < 0 ? '−' : '') +
            Math.floor(minutes / 480) +
            'd ' +
            Math.floor((minutes % 480) / 60) +
            'h ' +
            (minutes % 60) +
            'm',
    };
}
export default function moneyBalance() {
    return {
        eligible: true,
        used: 40,
        credit: true,
        pending: true,
        mix: false,
        get summary() {
            return ledgerSummary(this.mix);
        },
        get left() {
            return deriveBalance(this.used, this.credit, this.pending);
        },
    };
}
