<?php

namespace App\Livewire;

use App\Vault\BalancePricing;
use DomainException;
use Livewire\Component;

class CurrencyGuardBalance extends Component
{
    public bool $eligible = true;

    public bool $mix = false;

    public bool $credit = true;

    public bool $pending = true;

    public float $used = 40;

    private const LEDGER = [['cents' => 12500050, 'currency' => 'EUR'], ['cents' => 42000000, 'currency' => 'USD']];

    public function balances(): array
    {
        return app(BalancePricing::class)->group(self::LEDGER);
    }

    public function mixError(): ?string
    {
        if (! $this->mix) {
            return null;
        }
        try {
            app(BalancePricing::class)->singleCurrencyTotal(self::LEDGER);
        } catch (DomainException $error) {
            return $error->getMessage();
        }

        return null;
    }

    public function hours(): float
    {
        return app(BalancePricing::class)->remainingHours(120, $this->used, $this->credit ? 8 : 0, $this->pending ? 16 : 0);
    }

    public function render()
    {
        return view('livewire.currency-guard-balance');
    }
}

<?php

namespace App\Vault;

use DomainException;

final class BalancePricing
{
    /** Return integer minor-unit balances keyed by currency. */
    public function group(array $ledger): array
    {
        $totals = [];
        foreach ($ledger as $row) {
            $code = Currency::parse($row['currency'] ?? null)->value;
            $totals[$code] = ($totals[$code] ?? 0) + $row['cents'];
        }

        return $totals;
    }

    public function singleCurrencyTotal(array $ledger): int
    {
        $totals = $this->group($ledger);
        if (count($totals) !== 1) {
            throw new DomainException('Select a single currency.');
        }

        return array_values($totals)[0];
    }

    public function remainingHours(float $deserved, float $used, float $overtimeCredit, float $pending): float
    {
        return $deserved + $overtimeCredit - $used - $pending;
    }
}

<?php

namespace App\Vault;

use DomainException;

enum Currency: string
{
    case USD = 'USD';
    case EUR = 'EUR';

    public static function parse(?string $code): self
    {
        return self::tryFrom(strtoupper(trim($code ?? '')))
            ?? throw new DomainException('A supported currency code is required.');
    }
}
