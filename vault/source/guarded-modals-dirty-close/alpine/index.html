import { isNormalizedDirty } from '../../../lib/kernels/dirtyGuard';
import { normalizeSchedule, scheduleGuard } from './public/vault/schedule-guard.js';

type Mode = 'edit' | 'reset' | 'note';
const dialog = () => document.querySelector<HTMLDialogElement>('#guard-modal');

export default function createGuards() {
    let navigation: ReturnType<typeof scheduleGuard> | undefined;
    let mounted = true;
    return {
        baseline: normalizeSchedule('5', ['1', '3'], '09:00'),
        hours: '5',
        days: ['1', '3'],
        startTime: '09:00',
        pendingFile: false,
        pending: false,
        rejectNext: false,
        error: '',
        mode: 'edit' as Mode,
        word: '',
        note: '',
        discard: false,
        message: '',
        revision: 1,
        init() {
            mounted = true;
            navigation = scheduleGuard({
                dirty: () => this.dirty,
                pending: () => this.pending,
                requestDiscard: () => {
                    this.discard = true;
                },
                close: (proceed) => this.close(proceed),
            });
        },
        destroy() {
            mounted = false;
            navigation?.destroy();
        },
        get dirty() {
            return (
                this.pendingFile ||
                isNormalizedDirty(
                    this.baseline,
                    normalizeSchedule(this.hours, this.days, this.startTime),
                )
            );
        },
        restore() {
            this.hours = this.baseline.hours;
            this.days = [...this.baseline.days];
            this.startTime = this.baseline.startTime;
            this.pendingFile = false;
            const file = dialog()?.querySelector<HTMLInputElement>('input[type=file]');
            if (file) file.value = '';
            this.rejectNext = false;
            this.error = '';
            this.discard = false;
        },
        open(mode: Mode) {
            this.mode = mode;
            this.word = '';
            this.note = '';
            this.restore();
            if (mode === 'note') {
                const popup = document.querySelector<HTMLElement>('#guard-note');
                popup?.showPopover();
                popup?.querySelector('textarea')?.focus();
                return;
            }
            dialog()?.showModal();
            if (mode === 'edit') navigation?.begin();
            requestAnimationFrame(() =>
                dialog()?.querySelector<HTMLElement>('[data-cancel]')?.focus(),
            );
        },
        toggle(day: string) {
            this.days = this.days.includes(day)
                ? this.days.filter((d) => d !== day)
                : [...this.days, day];
        },
        stage(files: FileList | null) {
            this.pendingFile = Boolean(files?.length);
        },
        close(proceed = false) {
            if (this.mode === 'note') {
                document.querySelector<HTMLElement>('#guard-note')?.hidePopover();
                document.querySelector<HTMLButtonElement>('[data-note-trigger]')?.focus();
            } else {
                if (this.mode === 'edit') navigation?.end(proceed);
                dialog()?.close();
            }
            this.restore();
        },
        keepEditing() {
            navigation?.stay();
            this.discard = false;
            requestAnimationFrame(() =>
                dialog()?.querySelector<HTMLElement>('[data-hours]')?.focus(),
            );
        },
        cancel() {
            if (this.pending) return;
            if (this.discard) this.keepEditing();
            else if (this.mode === 'edit' && this.dirty) this.discard = true;
            else this.close();
        },
        async save() {
            if (this.pending) return;
            if (this.mode === 'edit') {
                const draft = normalizeSchedule(this.hours, this.days, this.startTime);
                this.pending = true;
                const reject = this.rejectNext;
                this.rejectNext = false;
                this.error = '';
                // Simulate a request acknowledgement locally; attachments never enter its payload.
                await new Promise<void>((resolve) => setTimeout(resolve, 180));
                if (!mounted) return;
                this.pending = false;
                if (reject)
                    this.error = 'The demo service rejected this save. Your edits are retained.';
                else if (!/^\d+(?:\.\d+)?$/.test(draft.hours) || Number(draft.hours) > 24)
                    this.error = 'Enter hours between 0 and 24.';
                else if (!/^(?:[01]\d|2[0-3]):[0-5]\d$/.test(draft.startTime))
                    this.error = 'Enter a start time in 24-hour format, such as 09:00.';
                if (this.error) return;
                this.baseline = draft;
                this.message = 'Schedule saved.';
            } else if (this.mode === 'reset') {
                if (this.word !== 'RESET') return;
                this.message = 'Demo credentials reset to revision ' + ++this.revision + '.';
            } else {
                if (!this.note.trim()) return;
                this.message = 'Decision recorded: ' + this.note.trim();
            }
            this.close();
        },
    };
}

src/lib/kernels/dirtyGuard.ts

// Form values are JSON scalars, nested field records, or arrays of selected IDs.
function fieldKey(value: unknown): string {
    if (Array.isArray(value)) {
        return 'set:' + JSON.stringify([...new Set(value.map(fieldKey))].sort());
    }
    if (value !== null && typeof value === 'object') {
        return (
            'record:' +
            JSON.stringify(
                Object.entries(value)
                    .sort(([left], [right]) => (left < right ? -1 : left > right ? 1 : 0))
                    .map(([key, field]) => [key, fieldKey(field)]),
            )
        );
    }
    if (value === null || typeof value === 'string' || typeof value === 'number') {
        const normalized = String(value ?? '').trim().replace(/^((?:[01]\d|2[0-3]):[0-5]\d):00$/, '$1');
        return 'field:' + normalized;
    }
    return typeof value + ':' + String(value);
}

export function isNormalizedDirty(baseline: unknown, current: unknown): boolean {
    return fieldKey(baseline) !== fieldKey(current);
}

/**
 * The editor owns one same-document history entry. Back dismisses it; a dirty editor asks first.
 * Cross-document navigation and reload use the browser's native beforeunload confirmation.
 * @param {{dirty:()=>boolean,pending:()=>boolean,requestDiscard:()=>void,close:(proceed?:boolean)=>void}} callbacks
 */
export function scheduleGuard(callbacks) {
    const key = '__vaultScheduleEditor';
    const token = crypto.randomUUID?.() ?? String(performance.timeOrigin + performance.now());
    let active = false;
    /** @type {string|null} */
    let destination = null;
    const marked = () => history.state?.[key] === token;
    function push() {
        history.pushState({ ...history.state, [key]: token }, '', location.href);
    }
    /** @param {BeforeUnloadEvent} event */
    function beforeUnload(event) {
        if (active && (callbacks.dirty() || callbacks.pending())) {
            event.preventDefault();
        }
    }
    /** @param {MouseEvent} event */
    function linkClick(event) {
        if (
            !active ||
            event.defaultPrevented ||
            event.button !== 0 ||
            event.metaKey ||
            event.ctrlKey ||
            event.altKey ||
            event.shiftKey
        )
            return;
        const link = event.target instanceof Element ? event.target.closest('a[href]') : null;
        if (
            !(link instanceof HTMLAnchorElement) ||
            link.download ||
            (link.target && link.target !== '_self')
        )
            return;
        event.preventDefault();
        event.stopImmediatePropagation();
        if (callbacks.pending()) return;
        destination = link.href;
        if (callbacks.dirty()) callbacks.requestDiscard();
        else callbacks.close(true);
    }
    function pop() {
        if (!active || marked()) return;
        if (callbacks.dirty() || callbacks.pending()) {
            push();
            if (!callbacks.pending()) {
                destination = null;
                callbacks.requestDiscard();
            }
        } else callbacks.close();
    }
    function unbind() {
        window.removeEventListener('beforeunload', beforeUnload);
        window.removeEventListener('popstate', pop);
        document.removeEventListener('click', linkClick, true);
    }
    return {
        begin() {
            if (active) return;
            active = true;
            destination = null;
            push();
            window.addEventListener('popstate', pop);
            window.addEventListener('beforeunload', beforeUnload);
            document.addEventListener('click', linkClick, true);
        },
        stay() {
            destination = null;
        },
        end(proceed = false) {
            if (!active) return;
            active = false;
            unbind();
            const href = proceed ? destination : null;
            destination = null;
            if (marked()) {
                if (href)
                    window.addEventListener('popstate', () => location.assign(href), {
                        once: true,
                    });
                history.back();
            } else if (href) location.assign(href);
        },
        destroy() {
            active = false;
            destination = null;
            unbind();
        },
    };
}

/** Normalize scalar form values without losing significant leading zeroes.
 * @param {string|number} hours
 * @param {(string|number)[]} days
 * @param {string} startTime
 * @returns {{hours:string,days:string[],startTime:string}}
 */
export function normalizeSchedule(hours, days, startTime) {
    return {
        hours: String(hours ?? '').trim(),
        days: [...new Set(days.map(String))].sort(),
        startTime: String(startTime ?? '')
            .trim()
            .replace(/^((?:[01]\d|2[0-3]):[0-5]\d):00$/, '$1'),
    };
}
