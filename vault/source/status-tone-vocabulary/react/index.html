import { useState } from 'react';
import { states, readPriority, writePriority, ratings } from './model';

export default function Vocabulary() {
    const [held, setHeld] = useState(true),
        [rating, setRating] = useState('3');
    const [raw, setRaw] = useState('urgent');
    const [saved, setSaved] = useState('Medium');
    const [message, setMessage] = useState('');
    function save() {
        try {
            setSaved(writePriority(raw));
            setMessage('Saved ' + raw);
        } catch (error) {
            setMessage((error as Error).message);
        }
    }
    return (
        <section className="kit-panel">
            <div className="kit-grid">
                {states.map((state) => (
                    <div className="kit-cell" key={state.label}>
                        <span className={'status-badge ' + state.shape} data-tone={state.tone}>
                            <span aria-hidden="true">{state.glyph}</span>
                            {state.label}
                        </span>
                        {state.label === 'In progress' && held && (
                            <span className="status-badge outline" data-tone="warn">
                                Ⅱ On hold
                            </span>
                        )}
                        {state.urgency && (
                            <small data-urgency={state.urgency.variant}>
                                {state.urgency.label}
                            </small>
                        )}
                    </div>
                ))}
            </div>
            <label>
                <input type="checkbox" checked={held} onChange={(e) => setHeld(e.target.checked)} />{' '}
                Place in-progress work on hold
            </label>
            <label>
                Risk rating{' '}
                <select value={rating} onChange={(e) => setRating(e.target.value)}>
                    {['', '1', '2', '3'].map((v) => (
                        <option key={v} value={v}>
                            {v || 'Not set'}
                        </option>
                    ))}
                </select>
            </label>
            <p>Likelihood: {ratings[rating]}</p>
            <label>
                Priority input
                <input value={raw} onChange={(e) => setRaw(e.target.value)} />
            </label>
            <p>
                Reader <strong>{readPriority(raw)}</strong>
            </p>
            <div className="kit-row">
                <button onClick={save}>Write priority</button>
                <span>
                    Stored <b>{saved}</b>
                </span>
            </div>
            <output aria-live="polite">{message}</output>
        </section>
    );
}

import { rankPriority } from '../../../lib/kernels/sla';

export const toneMap = {
    work: { label: 'In progress', shape: 'solid', glyph: '●', tone: 'accent' },
    done: { label: 'Approved', shape: 'surface', glyph: '●', tone: 'ok' },
    attn: { label: 'Needs review', shape: 'outline', glyph: '◐', tone: 'warn' },
    stop: { label: 'Blocked', shape: 'dashed', glyph: '■', tone: 'crit' },
    wait: { label: 'Waiting', shape: 'hairline', glyph: '○', tone: 'neutral' },
} as const;

const urgencyMap = {
    act: { variant: 'act', label: 'Active' },
    late: { variant: 'late', label: 'Overdue' },
    rush: { variant: 'rush', label: 'Starts in 2d' },
    soft: { variant: 'soft', label: 'Scheduled' },
} as const;

export const states = [
    { ...toneMap.work, urgency: urgencyMap.act },
    { ...toneMap.done, urgency: null },
    { ...toneMap.attn, urgency: urgencyMap.rush },
    { ...toneMap.stop, urgency: urgencyMap.late },
    { ...toneMap.wait, urgency: urgencyMap.soft },
];
export const ratings: Record<string, string> = {
    '1': 'Low',
    '2': 'Medium',
    '3': 'High',
    '': 'Not rated',
};
const priorities = ['Critical', 'High', 'Medium', 'Low'];
export const readPriority = (raw: string) =>
    priorities[rankPriority(raw)] ?? 'Unknown · sorted last';
export function writePriority(raw: string) {
    if (!priorities.includes(raw)) throw new Error('Choose Critical, High, Medium or Low.');
    return raw;
}
export default function vocabulary() {
    return {
        states,
        held: true,
        rating: '3',
        ratings,
        readPriority,
        raw: 'urgent',
        saved: 'Medium',
        message: '',
        save() {
            try {
                this.saved = writePriority(this.raw);
                this.message = 'Saved ' + this.saved;
            } catch (error) {
                this.message = (error as Error).message;
            }
        },
    };
}

src/lib/kernels/sla.ts

type Instant = string | number;
export interface SlaTicket {
    created_at: Instant | null;
    sla_due_at: Instant | null;
    resolved_at?: Instant | null;
    status?: string;
    priority?: string;
}

const hour = 3_600_000;
const priority = ['critical', 'high', 'medium', 'low'];
const stamp = (time: Instant | null | undefined) =>
    typeof time === 'number' ? time : time ? Date.parse(time) : NaN;
const closed = (ticket: SlaTicket) =>
    ['closed', 'resolved'].includes(ticket.status?.toLowerCase() ?? '');

export function slaState(ticket: SlaTicket, now: Instant) {
    if (closed(ticket)) return 'closed';
    const left = stamp(ticket.sla_due_at) - stamp(now);
    if (!Number.isFinite(left)) return 'unknown';
    if (left < 0) return 'breached';
    return left < 2 * hour ? 'closing' : 'ok';
}

export function slaElapsedPercent(ticket: SlaTicket, now: Instant): number | null {
    const start = stamp(ticket.created_at);
    const budget = stamp(ticket.sla_due_at) - start;
    const elapsed = stamp(closed(ticket) ? ticket.resolved_at : now) - start;
    if (!Number.isFinite(elapsed) || !Number.isFinite(budget) || budget <= 0) return null;
    return Math.min(100, Math.max(2, (elapsed * 100) / budget));
}

export function rankPriority(raw: string | undefined): number {
    const value = raw?.trim().toLowerCase();
    if (value === 'urgent') return 0;
    const index = priority.indexOf(value ?? '');
    return index < 0 ? priority.length : index;
}

export function compareUrgency(a: SlaTicket, b: SlaTicket, now: Instant): number {
    const rank = { breached: 0, closing: 1, ok: 2, unknown: 3, closed: 4 };
    return (
        rank[slaState(a, now)] - rank[slaState(b, now)] ||
        stamp(a.sla_due_at) - stamp(b.sla_due_at) ||
        rankPriority(a.priority) - rankPriority(b.priority) ||
        stamp(a.created_at) - stamp(b.created_at) ||
        0
    );
}

export function durationParts(milliseconds: number) {
    const minutes = Math.floor(Math.abs(milliseconds) / 60_000);
    return { hours: Math.floor(minutes / 60), minutes: minutes % 60 };
}
