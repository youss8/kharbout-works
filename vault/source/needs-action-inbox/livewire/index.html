<?php

namespace App\Livewire;

use Livewire\Attributes\Locked;
use Livewire\Component;

class NeedsActionInbox extends Component
{
    #[Locked]
    public array $items = [
        'leave:81' => ['Annual leave · three days', 1, null, 3, 'Backup coordinator also away 28–30 December'],
        'attendance:28' => ['Attendance correction', 1, null, 2, ''],
        'contract:13' => ['Contract change review', 1, null, 1, ''],
    ];

    #[Locked]
    public ?array $undoItem = null;

    public bool $fault = false;

    public string $outcome = '';

    public function stale(string $key): void
    {
        if (isset($this->items[$key])) {
            $this->items[$key][1]++;
        }
    }

    /** @return array{outcome:'changed'|'already_handled'|'done'} */
    public function decide(string $key, int $version, string $decision): array
    {
        abort_unless(isset($this->items[$key]) && in_array($decision, ['Approve', 'Snooze'], true), 422);
        $result = WorkInbox::checkDecision($this->items[$key], $version);
        if ($result === InboxOutcome::Done) {
            $this->items[$key][1]++;
            $this->items[$key][2] = $decision;
            $this->undoItem = [$key, $this->items[$key][1], microtime(true) + 6];
        }
        $this->outcome = $result->value;

        return ['outcome' => $this->outcome];
    }

    public function undo(): void
    {
        if (! $this->undoItem) {
            return;
        }
        [$key,$version,$until] = $this->undoItem;
        $this->undoItem = null;
        if ($until <= microtime(true) || $this->items[$key][1] !== $version) {
            $this->outcome = 'changed';

            return;
        }
        $this->items[$key][1]++;
        $this->items[$key][2] = null;
        $this->outcome = 'Decision undone.';
    }

    public function render()
    {
        $digest = WorkInbox::digest($this->fault ? 'licenses' : null);

        return view('livewire.needs-action-inbox', compact('digest'));
    }
}

<?php

namespace App\Livewire;

use RuntimeException;

final class WorkInbox
{
    public static function digest(?string $unavailable): array
    {
        $adapters = [
            'loans' => ['Overdue loans', 'crit', fn () => count(array_filter([-4, -2, 3, -1, -7], fn ($d) => $d < 0))],
            'licenses' => ['Expiring licenses', 'warn', fn () => count(array_filter([5, 12, 50], fn ($d) => $d <= 30))],
            'returns' => ['Late site returns', 'crit', fn () => self::countState(['late', 'late', 'submitted', 'late'], 'late')],
            'access' => ['Access reviews', 'warn', fn () => self::countState(['complete', 'complete'], 'pending')],
            'leave' => ['Leave requests', 'warn', fn () => self::countState(['pending', 'approved', 'pending'], 'pending')],
            'contracts' => ['Ending contracts', 'warn', fn () => count(array_filter([9, 40, 83], fn ($d) => $d <= 30))],
        ];
        $rows = [];
        $failed = [];
        foreach ($adapters as $key => [$label,$urgency,$read]) {
            try {
                if ($unavailable === $key) {
                    throw new RuntimeException($label.' unavailable');
                }
                $count = $read();
                $tone = $count === 0 ? 'neutral' : $urgency;
                $glyph = ['neutral' => '○', 'crit' => '■', 'warn' => '◐'][$tone];
                $rows[] = compact('key', 'label', 'count', 'tone', 'glyph');
            } catch (RuntimeException $error) {
                $failed[$key] = $error->getMessage();
            }
        }

        return compact('rows', 'failed');
    }

    private static function countState(array $records, string $status): int
    {
        return count(array_filter($records, fn ($state) => $status === $state));
    }

    public static function checkDecision(array $row, int $expected): InboxOutcome
    {
        if ($row[2] !== null) {
            return InboxOutcome::AlreadyHandled;
        }

        return $row[1] === $expected ? InboxOutcome::Done : InboxOutcome::Changed;
    }
}

enum InboxOutcome: string
{
    case Changed = 'changed';
    case AlreadyHandled = 'already_handled';
    case Done = 'done';
}
