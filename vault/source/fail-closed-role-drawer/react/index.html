import { useKitModel } from '../../../components/vault/useKitModel';
import createDrawer from './model';
export default function Example() {
    const m = useKitModel(createDrawer);
    return (
        <section
            className="kit-panel"
            onKeyDown={(e) => {
                if (e.key === 'Escape') m.close();
            }}
        >
            <label>
                Active role{' '}
                <select value={m.role} onChange={(e) => m.change(e.target.value as typeof m.role)}>
                    <option>Regional HR</option>
                    <option>Manager</option>
                    <option>Employee</option>
                </select>
            </label>
            <div className="kit-row">
                {m.colleagues.map((p) => (
                    <button key={p.id} onClick={() => m.open(p.id)}>
                        {p.label}
                    </button>
                ))}
                <button onClick={() => m.open(404)}>Open missing record</button>
            </div>
            {m.requestedId !== null && (
                <aside aria-label="Profile detail" className="kit-panel profile-drawer">
                    <button onClick={() => m.close()}>Close detail</button>
                    <div data-loading={m.loading || undefined} aria-busy={m.loading}>
                        {m.loading && <p role="status">Checking access…</p>}
                        {m.ready && m.profile && (
                            <article>
                                <h3>{m.profile.label}</h3>
                                <p>{m.profile.assignment}</p>
                            </article>
                        )}
                        {m.unavailable && (
                            <div role="alert" className="error-card">
                                Record unavailable{' '}
                                <button onClick={() => m.open(m.requestedId!)}>Retry</button>
                            </div>
                        )}
                    </div>
                </aside>
            )}
        </section>
    );
}

src/components/vault/useKitModel.ts

import { useEffect, useState, useSyncExternalStore } from 'react';

/** Both preview stacks observe the same model. Replace arrays and records when updating them. */
export function useKitModel<T extends object>(create: () => T): T {
    const [store] = useState(() => {
        let version = 0;
        const listeners = new Set<() => void>();
        const model = new Proxy(create(), {
            set(target, key, value) {
                if (Object.is(Reflect.get(target, key), value)) return true;
                const written = Reflect.set(target, key, value);
                if (written) {
                    version++;
                    listeners.forEach((notify) => notify());
                }
                return written;
            },
        });
        return {
            model,
            version: () => version,
            subscribe(notify: () => void) {
                listeners.add(notify);
                return () => {
                    listeners.delete(notify);
                };
            },
        };
    });
    useSyncExternalStore(store.subscribe, store.version, store.version);
    useEffect(() => {
        const model = store.model as T & { init?(): void; destroy?(): void };
        model.init?.();
        return () => model.destroy?.();
    }, [store]);
    return store.model;
}

src/content/vault/fail-closed-role-drawer/model.ts

type Role = 'Regional HR' | 'Manager' | 'Employee';
type Profile = { id: number; label: string; assignment: string };
export const colleagues = [
    { id: 42, label: 'Staff 042' },
    { id: 88, label: 'Staff 088' },
    { id: 91, label: 'Staff 091' },
];
function permittedProfile(id: number, role: Role): Profile | null {
    const row = colleagues.find((p) => p.id === id);
    if (!row || (id === 88 && role !== 'Regional HR') || (id === 91 && role === 'Employee'))
        return null;
    return {
        ...row,
        assignment:
            id === 42
                ? 'Site coordination'
                : id === 88
                  ? 'Restricted contract review'
                  : 'Operations planning',
    };
}
export default function createDrawer() {
    let sequence = 0;
    return {
        colleagues,
        role: 'Regional HR' as Role,
        requestedId: null as number | null,
        loadedId: null as number | null,
        profile: null as Profile | null,
        loading: false,
        unavailable: false,
        get ready() {
            return (
                this.requestedId !== null &&
                this.loadedId === this.requestedId &&
                this.profile?.id === this.requestedId
            );
        },
        async open(id: number) {
            const token = ++sequence,
                context = this.role;
            this.requestedId = id;
            this.loadedId = null;
            this.profile = null;
            this.unavailable = false;
            this.loading = true;
            await new Promise((resolve) => setTimeout(resolve, id === 88 ? 540 : 100));
            if (token !== sequence || context !== this.role || this.requestedId !== id) return;
            const record = permittedProfile(id, context);
            this.loading = false;
            this.unavailable = record === null;
            this.profile = record;
            this.loadedId = record?.id ?? null;
        },
        change(role: Role) {
            this.role = role;
            if (this.requestedId !== null) void this.open(this.requestedId);
        },
        close() {
            sequence++;
            this.requestedId = null;
            this.loadedId = null;
            this.profile = null;
            this.loading = false;
            this.unavailable = false;
        },
        destroy() {
            sequence++;
        },
    };
}
